🚀 快速开始

# 克隆仓库
git clone git@github.com:Gip886/agents.git
cd agents

# 每天进入对应目录
cd day1
python3 -m venv venv
source venv/bin/activate
pip install -r requirements.txt

# 复制 .env.example 为 .env，填入你的 API Key
cp .env.example .env

# 跑 demo
python demo1_single_chat.py
